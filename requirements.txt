# Resume analyser / resume-to-job matching starter dependencies.
# Install: python -m pip install -r requirements.txt
# Versions are intentionally unpinned: this is a starter list, not a tested lockfile.

# AI analysis, structured extraction, and embeddings
openai                  # Official OpenAI Python SDK; requires OPENAI_API_KEY
langchain               # Compose model calls and analysis workflows
langchain-openai        # LangChain's OpenAI chat and embedding integration
langchain-text-splitters # Split long resumes and job descriptions into chunks
tiktoken                # Count tokens for OpenAI model inputs
pydantic                # Validate structured resume fields and analysis results
python-dotenv           # Load local configuration, including API keys, from .env

# Resume document parsing
langchain-community     # PyPDFLoader and UnstructuredWordDocumentLoader
pypdf>=6,<7             # Extract PDF text; image-only scans need OCR
python-docx>=1.2,<2      # Read DOCX paragraphs and tables in document order
unstructured[docx]      # Parse .docx files with UnstructuredWordDocumentLoader

# Desktop interface
PySide6>=6.8,<7          # Qt 6 desktop widgets; minimal setup: requirements-desktop.txt

# Fetch and parse job-posting pages
requests                # Download HTML and call job-listing APIs
beautifulsoup4          # Parse HTML; import with: from bs4 import BeautifulSoup
lxml                    # HTML/XML parser for Beautiful Soup
sentence-transformers   # Local embeddings in src/nlp/embeddings.py

# Data processing and matching (used by resume_match.ipynb)
numpy                   # Numerical arrays and embedding calculations
pandas                  # Organize resume/job data and matching results
scikit-learn            # TF-IDF and cosine similarity; import name: sklearn

# Existing notebook plots
matplotlib              # Plot matching results
seaborn                 # Statistical visualizations

# Optional: uncomment only the packages needed for your implementation.

# deepagents            # Optional agent experiments; not needed for HTML extraction
# playwright            # Render JavaScript-driven job pages

# scrapy                # Crawl many pages with scheduling and pipelines
# langchain-chroma      # Store/search embeddings through LangChain (installs chromadb)
# Local Gemma agent: install -r requirements-model.txt separately.
# Baidu PDF OCR: install requirements-ocr.txt in a separate .venv-ocr environment.
# streamlit             # Upload resumes and display results in a simple web UI
# ipykernel             # Run this environment as a Jupyter/VS Code notebook kernel
